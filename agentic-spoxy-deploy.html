// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/security/ReentrancyGuard.sol";

interface IAgentId {
    function balanceOf(address account) external view returns (uint256);
}

/// @title AGENTIC SPOXY
/// @notice ERC-721 on BNB Chain (chain 56). Mint only for wallets holding a Binance Agent ID.
/// Free for the first 100 tokens of total supply, then 0.00025 BNB per mint.
/// Max 2 tokens per wallet in total. Max supply 5000.
/// Owner can open/close mint, update baseURI, and withdraw.
/// BaseURI is settable so the real IPFS CID can be set after upload.
/// Initial baseURI is a pattern and MUST be updated by the owner after IPFS upload.
contract AgenticSpoxy is ERC721, Ownable, ReentrancyGuard {
    uint256 public constant MAX_SUPPLY = 5000;
    uint256 public constant FREE_SUPPLY = 100;
    uint256 public constant MINT_PRICE = 0.00025 ether;
    uint256 public constant MAX_PER_WALLET = 2;

    // Binance .ai Agent ID registry as used by LeadLoop-style minting on BNB Chain.
    // VERIFY this address before signing a mainnet deploy. If it is wrong,
    // minting will be gated by the wrong token. Owner cannot change it in this
    // version on purpose, so a wrong registry requires a redeploy.
    address public constant AGENT_ID_REGISTRY = 0x60B9262456dE011F7d15a54CdD8c48bAaa8B022F;

    uint256 public totalMinted;
    bool public mintOpen = true;
    string private _baseTokenURI = "ipfs://AGENTIC-SPOXY/";

    mapping(address => uint256) public mintedByWallet;

    error NoAgentId();
    error MintClosed();
    error BadQuantity();
    error OverLimit();
    error SoldOut();
    error Underpaid();

    constructor() ERC721("AGENTIC SPOXY", "ASPOXY") {}

    function mint(uint256 qty) external payable nonReentrant {
        if (!mintOpen) revert MintClosed();
        if (qty == 0 || qty > MAX_PER_WALLET) revert BadQuantity();
        if (mintedByWallet[msg.sender] + qty > MAX_PER_WALLET) revert OverLimit();
        if (totalMinted + qty > MAX_SUPPLY) revert SoldOut();
        if (IAgentId(AGENT_ID_REGISTRY).balanceOf(msg.sender) == 0) revert NoAgentId();

        uint256 paidCount = 0;
        for (uint256 i = 0; i < qty; i++) {
            if (totalMinted + i >= FREE_SUPPLY) {
                paidCount += 1;
            }
        }
        uint256 required = paidCount * MINT_PRICE;
        if (msg.value < required) revert Underpaid();

        mintedByWallet[msg.sender] += qty;
        for (uint256 i = 0; i < qty; i++) {
            totalMinted += 1;
            _safeMint(msg.sender, totalMinted);
        }

        uint256 refund = msg.value - required;
        if (refund > 0) {
            (bool ok, ) = payable(msg.sender).call{value: refund}("");
            require(ok, "Refund failed");
        }
    }

    function tokenURI(uint256 tokenId) public view override returns (string memory) {
        _requireMinted(tokenId);
        return string(abi.encodePacked(_baseTokenURI, _toString(tokenId), ".json"));
    }

    function _toString(uint256 value) internal pure returns (string memory) {
        if (value == 0) return "0";
        uint256 temp = value;
        uint256 digits;
        while (temp != 0) {
            digits++;
            temp /= 10;
        }
        bytes memory buffer = new bytes(digits);
        while (value != 0) {
            digits -= 1;
            buffer[digits] = bytes1(uint8(48 + uint256(value % 10)));
            value /= 10;
        }
        return string(buffer);
    }

    function _baseURI() internal view override returns (string memory) {
        return _baseTokenURI;
    }

    function setBaseURI(string calldata newBaseURI) external onlyOwner {
        _baseTokenURI = newBaseURI;
    }

    function setMintOpen(bool open) external onlyOwner {
        mintOpen = open;
    }

    function withdraw() external onlyOwner nonReentrant {
        (bool ok, ) = payable(owner()).call{value: address(this).balance}("");
        require(ok, "Withdraw failed");
    }
}
